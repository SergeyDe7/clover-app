Функция ИзвлечьНомерЗаказаCloverИзКомментария(Комментарий) Экспорт
	Текст = СокрЛП(Строка(Комментарий));
	Маркер = "Заказ Clover № ";
	Позиция = Найти(Текст, Маркер);
	Если Позиция = 0 Тогда
		Возврат "";
	КонецЕсли;
	Хвост = СокрЛП(Сред(Текст, Позиция + СтрДлина(Маркер)));
	// Номера Clover вида CL-260731-185736-019 (буквы, цифры, дефис).
	Номер = "";
	Допустимые = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-";
	Для Индекс = 1 По СтрДлина(Хвост) Цикл
		Символ = Сред(Хвост, Индекс, 1);
		Если Найти(Допустимые, Символ) = 0 Тогда
			Прервать;
		КонецЕсли;
		Номер = Номер + Символ;
	КонецЦикла;
	Возврат СокрЛП(Номер);
