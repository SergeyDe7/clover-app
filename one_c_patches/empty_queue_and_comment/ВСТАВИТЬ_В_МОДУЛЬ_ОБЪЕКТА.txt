&После("ПриЗаписи")
Процедура Clover_ПриЗаписи(Отказ)

	Если Отказ Тогда
		Возврат;
	КонецЕсли;

	Если ЭтотОбъект.ДополнительныеСвойства.Свойство("CloverSkipAcceptedNotify") Тогда
		Возврат;
	КонецЕсли;

	Комментарий = Строка(ЭтотОбъект.Комментарий);
	НомерClover = Clover_ОбменКлиент.ИзвлечьНомерЗаказаCloverИзКомментария(Комментарий);
	Если НомерClover = "" Тогда
		Возврат;
	КонецЕсли;

	СостояниеТекст = "";
	Попытка
		СостояниеТекст = СокрЛП(Строка(ЭтотОбъект.СостояниеЗаказа));
	Исключение
		Попытка
			СостояниеТекст = СокрЛП(Строка(ЭтотОбъект.Состояние));
		Исключение
			СостояниеТекст = "";
		КонецПопытки;
	КонецПопытки;

	Если СостояниеТекст = "" Тогда
		Возврат;
	КонецЕсли;

	Попытка
		Clover_ОбменКлиент.СообщитьОПринятииПоРеквизитам(
			НомерClover,
			Строка(ЭтотОбъект.Номер),
			СостояниеТекст);
	Исключение
		ЗаписьЖурналаРегистрации(
			"Clover.AcceptedNotify",
			УровеньЖурналаРегистрации.Ошибка,,,
			ОписаниеОшибки());
	КонецПопытки;

КонецПроцедуры
