Функция Clover_HTTP_JSON(Метод, Путь, ТелоСтрока = "")
	
	Результат = Новый Структура("КодСостояния, Тело, Ошибка", -1, "", "");
	
	Ключ = Clover_КлючОбменаAPI();
	Если ПустаяСтрока(Ключ) Тогда
		Результат.Ошибка = "Не заполнен Clover_КлючОбменаAPI() — скопируйте ключ из ОтправитьКонтрагентыПредпросмотр.";
		Возврат Результат;
	КонецЕсли;
	
	Попытка
		Соединение = Новый HTTPСоединение(Clover_ХостAPI(), Clover_ПортAPI(), , , , 60);
		Запрос = Новый HTTPЗапрос(Путь);
		Запрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
		Запрос.Заголовки.Вставить("X-Clover-Key", Ключ);
		Запрос.Заголовки.Вставить("X-Clover-Database", "TEST");
		Если ЗначениеЗаполнено(ТелоСтрока) Тогда
			Запрос.УстановитьТелоИзСтроки(ТелоСтрока, КодировкаТекста.UTF8);
		КонецЕсли;
		
		Если ВРег(Метод) = "GET" Тогда
			Ответ = Соединение.Получить(Запрос);
		Иначе
			Ответ = Соединение.ОтправитьДляОбработки(Запрос);
		КонецЕсли;
		
		Результат.КодСостояния = Ответ.КодСостояния;
		Результат.Тело = Ответ.ПолучитьТелоКакСтроку(КодировкаТекста.UTF8);
	Исключение
		Результат.Ошибка = ОписаниеОшибки();
	КонецПопытки;
	
	Возврат Результат;
	
КонецФункции

Функция Clover_ВидЦенИзДоговора(КонтрагентСсылка)
	
	Пустой = Новый Структура("id, name, code", "", "", "");
	Если Не ЗначениеЗаполнено(КонтрагентСсылка) Тогда
		Возврат Пустой;
	КонецЕсли;
	
	ВидЦенСсылка = Неопределено;
	
	// 1) Договор по умолчанию у контрагента (если реквизит есть)
	Попытка
		ДоговорПоУмолчанию = КонтрагентСсылка.ДоговорПоУмолчанию;
		Если ЗначениеЗаполнено(ДоговорПоУмолчанию) Тогда
			ВидЦенСсылка = ДоговорПоУмолчанию.ВидЦен;
		КонецЕсли;
	Исключение
	КонецПопытки;
	
	// 2) Первый действующий договор контрагента
	Если Не ЗначениеЗаполнено(ВидЦенСсылка) Тогда
		Попытка
			Запрос = Новый Запрос;
			Запрос.Текст =
			"ВЫБРАТЬ ПЕРВЫЕ 1
			|	Договоры.ВидЦен КАК ВидЦен
			|ИЗ
			|	Справочник.ДоговорыКонтрагентов КАК Договоры
			|ГДЕ
			|	Договоры.Владелец = &Контрагент
			|	И НЕ Договоры.ПометкаУдаления
			|	И Договоры.ВидЦен <> ЗНАЧЕНИЕ(Справочник.ВидыЦен.ПустаяСсылка)";
			Запрос.УстановитьПараметр("Контрагент", КонтрагентСсылка);
			Выборка = Запрос.Выполнить().Выбрать();
			Если Выборка.Следующий() Тогда
				ВидЦенСсылка = Выборка.ВидЦен;
			КонецЕсли;
		Исключение
			// Запасной вариант имени справочника видов цен / договоров — см. инструкцию
		КонецПопытки;
	КонецЕсли;
	
	Если Не ЗначениеЗаполнено(ВидЦенСсылка) Тогда
		Возврат Пустой;
	КонецЕсли;
	
	Результат = Новый Структура;
	Результат.Вставить("id", Строка(ВидЦенСсылка.УникальныйИдентификатор()));
	Результат.Вставить("name", Строка(ВидЦенСсылка));
	Код = "";
	Попытка
		Код = ВидЦенСсылка.Код;
	Исключение
	КонецПопытки;
	Результат.Вставить("code", Код);
	Возврат Результат;
	
КонецФункции

// Вызов из меню Clover (новая кнопка) или вручную из отладки.
Функция ОтправитьВидыЦенВClover() Экспорт
	
	Элементы = Новый Массив;
	
	Попытка
		Запрос = Новый Запрос;
		Запрос.Текст =
		"ВЫБРАТЬ
		|	ВидыЦен.Ссылка КАК Ссылка,
		|	ВидыЦен.Код КАК Код,
		|	ВидыЦен.Наименование КАК Наименование
		|ИЗ
		|	Справочник.ВидыЦен КАК ВидыЦен
		|ГДЕ
		|	НЕ ВидыЦен.ПометкаУдаления
		|УПОРЯДОЧИТЬ ПО
		|	ВидыЦен.Наименование";
		Выборка = Запрос.Выполнить().Выбрать();
		Пока Выборка.Следующий() Цикл
			Элемент = Новый Структура;
			Элемент.Вставить("id", Строка(Выборка.Ссылка.УникальныйИдентификатор()));
			Элемент.Вставить("code", Выборка.Код);
			Элемент.Вставить("name", Выборка.Наименование);
			Элементы.Добавить(Элемент);
		КонецЦикла;
	Исключение
		// Если справочник называется иначе (ВидыЦенНоменклатуры) — поправьте текст запроса.
		Возврат Новый Структура("КодСостояния, Количество, Ошибка", -1, 0, ОписаниеОшибки());
	КонецПопытки;
	
	Данные = Новый Структура;
	Данные.Вставить("items", Элементы);
	
	Запись = Новый ЗаписьJSON;
	Запись.УстановитьСтроку();
	ЗаписатьJSON(Запись, Данные);
	Тело = Запись.Закрыть();
	
	Ответ = Clover_HTTP_JSON("POST", "/api/one-c/price-types", Тело);
	Возврат Новый Структура("КодСостояния, Количество, Ошибка, Тело",
		Ответ.КодСостояния, Элементы.Количество(), Ответ.Ошибка, Ответ.Тело);
	
КонецФункции

Функция ОтправитьПродажныеЦеныВClover() Экспорт
	
	ЗапросСписка = Clover_HTTP_JSON("GET", "/api/one-c/sale-price-request", "");
	Если ЗапросСписка.КодСостояния <> 200 Тогда
		Возврат Новый Структура("КодСостояния, Количество, ПропущеноНулей, Ошибка, Тело",
			ЗапросСписка.КодСостояния, 0, 0, ЗапросСписка.Ошибка, ЗапросСписка.Тело);
	КонецЕсли;
	
	Чтение = Новый ЧтениеJSON;
	Чтение.УстановитьСтроку(ЗапросСписка.Тело);
	ДанныеЗапроса = ПрочитатьJSON(Чтение);
	Чтение.Закрыть();
	
	Нужные = Новый Массив;
	Если ТипЗнч(ДанныеЗапроса) = Тип("Структура") И ДанныеЗапроса.Свойство("items") Тогда
		Нужные = ДанныеЗапроса.items;
	КонецЕсли;
	
	Если Нужные.Количество() = 0 Тогда
		Возврат Новый Структура("КодСостояния, Количество, ПропущеноНулей, Ошибка, Тело",
			200, 0, 0, "Нет клиентов с видом цен / нечего выгружать", "");
	КонецЕсли;
	
	Пакет = Новый Массив;
	ПропущеноНулей = 0;
	
	Для Каждого СтрокаНужна Из Нужные Цикл
		
		ИдНоменклатуры = "";
		ИдВидаЦен = "";
		Если ТипЗнч(СтрокаНужна) = Тип("Структура") Тогда
			Если СтрокаНужна.Свойство("id") Тогда
				ИдНоменклатуры = Строка(СтрокаНужна.id);
			КонецЕсли;
			Если СтрокаНужна.Свойство("priceTypeId") Тогда
				ИдВидаЦен = Строка(СтрокаНужна.priceTypeId);
			КонецЕсли;
		КонецЕсли;
		Если ПустаяСтрока(ИдНоменклатуры) Или ПустаяСтрока(ИдВидаЦен) Тогда
			Продолжить;
		КонецЕсли;
		
		Попытка
			НоменклатураСсылка = Справочники.Номенклатура.ПолучитьСсылку(Новый УникальныйИдентификатор(ИдНоменклатуры));
			ВидЦенСсылка = Справочники.ВидыЦен.ПолучитьСсылку(Новый УникальныйИдентификатор(ИдВидаЦен));
		Исключение
			Продолжить;
		КонецПопытки;
		
		ЦенаЗначение = 0;
		Попытка
			ЗапросЦены = Новый Запрос;
			ЗапросЦены.Текст =
			"ВЫБРАТЬ ПЕРВЫЕ 1
			|	Цены.Цена КАК Цена
			|ИЗ
			|	РегистрСведений.ЦеныНоменклатуры.СрезПоследних(
			|		,
			|		Номенклатура = &Номенклатура
			|			И ВидЦен = &ВидЦен) КАК Цены";
			ЗапросЦены.УстановитьПараметр("Номенклатура", НоменклатураСсылка);
			ЗапросЦены.УстановитьПараметр("ВидЦен", ВидЦенСсылка);
			ВыборкаЦены = ЗапросЦены.Выполнить().Выбрать();
			Если ВыборкаЦены.Следующий() Тогда
				ЦенаЗначение = ВыборкаЦены.Цена;
			КонецЕсли;
		Исключение
			Продолжить;
		КонецПопытки;
		
		// By design: нули из регистра не отправляем в Clover, только считаем.
		Если ЦенаЗначение = 0 Тогда
			ПропущеноНулей = ПропущеноНулей + 1;
			Продолжить;
		КонецЕсли;
		
		СтрокаПакета = Новый Структура;
		СтрокаПакета.Вставить("id", ИдНоменклатуры);
		СтрокаПакета.Вставить("priceTypeId", ИдВидаЦен);
		СтрокаПакета.Вставить("price", ЦенаЗначение);
		Пакет.Добавить(СтрокаПакета);
		
	КонецЦикла;
	
	Данные = Новый Структура;
	Данные.Вставить("items", Пакет);
	Запись = Новый ЗаписьJSON;
	Запись.УстановитьСтроку();
	ЗаписатьJSON(Запись, Данные);
	Тело = Запись.Закрыть();
	
	Ответ = Clover_HTTP_JSON("POST", "/api/one-c/sale-prices", Тело);
	Возврат Новый Структура("КодСостояния, Количество, ПропущеноНулей, Ошибка, Тело",
		Ответ.КодСостояния, Пакет.Количество(), ПропущеноНулей, Ответ.Ошибка, Ответ.Тело);
	
КонецФункции
