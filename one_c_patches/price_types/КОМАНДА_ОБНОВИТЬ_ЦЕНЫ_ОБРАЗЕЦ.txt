Процедура ОбработкаКоманды(ПараметрКоманды, ПараметрыВыполненияКоманды)
	
	// Важно: вызывать с клиента, НЕ через &НаСервере —
	// иначе 1С может заблокировать HTTPСоединение.
	//
	// Только продажные цены по видам (матрица клиентов).
	// Контрагентов и справочник видов цен не трогает.
	
	Результат = Clover_ОбменКлиент.ОтправитьПродажныеЦеныВClover();
	
	Если Результат.КодСостояния = 200 Тогда
		Сообщить("Цены обновлены в Clover: " + Строка(Результат.Количество));
	ИначеЕсли Результат.КодСостояния = -1 Тогда
		ТекстОшибки = "";
		Если Результат.Свойство("Ошибка") Тогда
			ТекстОшибки = Строка(Результат.Ошибка);
		КонецЕсли;
		Если ТекстОшибки = "" Тогда
			Сообщить("Не удалось обновить цены. Проверьте, что Clover API запущен.");
		Иначе
			Сообщить("Ошибка обновления цен: " + ТекстОшибки);
		КонецЕсли;
	Иначе
		Сообщить("Clover код " + Строка(Результат.КодСостояния)
			+ ". Цен в пакете: " + Строка(Результат.Количество)
			+ ?(Результат.Свойство("Ошибка") И СокрЛП(Строка(Результат.Ошибка)) <> "",
				". " + Строка(Результат.Ошибка),
				""));
	КонецЕсли;
