Процедура ОбработкаКоманды(ПараметрКоманды, ПараметрыВыполненияКоманды)
	
	Результат = Clover_ОбменКлиент.ОтправитьПродажныеЦеныВClover();
	
	ПропущеноНулей = 0;
	Если ТипЗнч(Результат) = Тип("Структура") И Результат.Свойство("ПропущеноНулей") Тогда
		ПропущеноНулей = Результат.ПропущеноНулей;
	КонецЕсли;
	
	Если Результат.КодСостояния = 200 Тогда
		Сообщить(
			"Продажные цены переданы в Clover: " + Строка(Результат.Количество)
			+ "; пропущено нулевых: " + Строка(ПропущеноНулей));
	ИначеЕсли Результат.КодСостояния = -1 Тогда
		Сообщить("Ошибка: " + Строка(Результат.Ошибка));
	Иначе
		Сообщить("Clover код " + Строка(Результат.КодСостояния) + " / " + Строка(Результат.Ошибка));
	КонецЕсли;
