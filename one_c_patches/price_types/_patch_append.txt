Функция Clover_ВидЦенИзДоговора(КонтрагентСсылка) Экспорт

	Пустой = Новый Структура("id, name, code", "", "", "");
	Если Не ЗначениеЗаполнено(КонтрагентСсылка) Тогда
		Возврат Пустой;
	КонецЕсли;

	ВидЦенСсылка = Неопределено;

	Попытка
		ДоговорПоУмолчанию = КонтрагентСсылка.ДоговорПоУмолчанию;
		Если ЗначениеЗаполнено(ДоговорПоУмолчанию) Тогда
			ВидЦенСсылка = ДоговорПоУмолчанию.ВидЦен;
		КонецЕсли;
	Исключение
	КонецПопытки;

	Если Не ЗначениеЗаполнено(ВидЦенСсылка) Тогда
		Попытка
			Запрос = Новый Запрос;
			Запрос.Текст =
			"ВЫБРАТЬ ПЕРВЫЕ 1
			|	Договоры.ВидЦен КАК ВидЦен
			|ИЗ
			|	Справочник.ДоговорыКонтрагентов КАК Договоры
			|ГДЕ
			|	Договоры.Владелец = &Контрагент
			|	И НЕ Договоры.ПометкаУдаления";
			Запрос.УстановитьПараметр("Контрагент", КонтрагентСсылка);
			Выборка = Запрос.Выполнить().Выбрать();
			Если Выборка.Следующий() И ЗначениеЗаполнено(Выборка.ВидЦен) Тогда
				ВидЦенСсылка = Выборка.ВидЦен;
			КонецЕсли;
		Исключение
		КонецПопытки;
	КонецЕсли;

	Если Не ЗначениеЗаполнено(ВидЦенСсылка) Тогда
		Возврат Пустой;
	КонецЕсли;

	Результат = Новый Структура;
	Результат.Вставить("id", Строка(ВидЦенСсылка.УникальныйИдентификатор()));
	Результат.Вставить("name", Строка(ВидЦенСсылка));
	Код = "";
	Попытка
		Код = ВидЦенСсылка.Код;
	Исключение
	КонецПопытки;
	Результат.Вставить("code", Код);
	Возврат Результат;

КонецФункции

Функция ОтправитьВидыЦенВClover() Экспорт

	Результат = Новый Структура;
	Результат.Вставить("КодСостояния", -1);
	Результат.Вставить("Количество", 0);
	Результат.Вставить("Ошибка", "");

	Попытка
		Элементы = Новый Массив;
		Запрос = Новый Запрос;
		Запрос.Текст =
		"ВЫБРАТЬ
		|	ВидыЦен.Ссылка КАК Ссылка,
		|	ВидыЦен.Код КАК Код,
		|	ВидыЦен.Наименование КАК Наименование
		|ИЗ
		|	Справочник.ВидыЦен КАК ВидыЦен
		|ГДЕ
		|	НЕ ВидыЦен.ПометкаУдаления
		|УПОРЯДОЧИТЬ ПО
		|	ВидыЦен.Наименование";
		Выборка = Запрос.Выполнить().Выбрать();
		Пока Выборка.Следующий() Цикл
			Элемент = Новый Структура;
			Элемент.Вставить("id", Строка(Выборка.Ссылка.УникальныйИдентификатор()));
			Элемент.Вставить("code", Выборка.Код);
			Элемент.Вставить("name", Выборка.Наименование);
			Элементы.Добавить(Элемент);
		КонецЦикла;

		Данные = Новый Структура;
		Данные.Вставить("items", Элементы);
		ЗаписьJSON = Новый ЗаписьJSON;
		ЗаписьJSON.УстановитьСтроку();
		ЗаписатьJSON(ЗаписьJSON, Данные);
		ТелоJSON = ЗаписьJSON.Закрыть();

		Соединение = Новый HTTPСоединение("192.168.155.15", 4100, , , , 60);
		ЗапросHTTP = Новый HTTPЗапрос("/api/one-c/price-types");
		ЗапросHTTP.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
		ДобавитьЗаголовкиОбмена(ЗапросHTTP);
		ЗапросHTTP.УстановитьТелоИзСтроки(ТелоJSON, КодировкаТекста.UTF8);
		Ответ = Соединение.ОтправитьДляОбработки(ЗапросHTTP);

		Результат.КодСостояния = Ответ.КодСостояния;
		Результат.Количество = Элементы.Количество();
		Если Ответ.КодСостояния <> 200 Тогда
			Результат.Ошибка = Ответ.ПолучитьТелоКакСтроку();
		КонецЕсли;
		Возврат Результат;
	Исключение
		Результат.Ошибка = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
		Возврат Результат;
	КонецПопытки;

КонецФункции

Функция ОтправитьПродажныеЦеныВClover() Экспорт

	Результат = Новый Структура;
	Результат.Вставить("КодСостояния", -1);
	Результат.Вставить("Количество", 0);
	Результат.Вставить("Ошибка", "");

	Попытка
		Соединение = Новый HTTPСоединение("192.168.155.15", 4100, , , , 120);

		ЗапросТребований = Новый HTTPЗапрос("/api/one-c/sale-price-request");
		ДобавитьЗаголовкиОбмена(ЗапросТребований);
		ОтветТребований = Соединение.Получить(ЗапросТребований);
		ТекстТребований = ОтветТребований.ПолучитьТелоКакСтроку();
		Если ОтветТребований.КодСостояния <> 200 Тогда
			Результат.КодСостояния = ОтветТребований.КодСостояния;
			Результат.Ошибка = ТекстТребований;
			Возврат Результат;
		КонецЕсли;

		ЧтениеJSON = Новый ЧтениеJSON;
		ЧтениеJSON.УстановитьСтроку(ТекстТребований);
		ДанныеТребований = ПрочитатьJSON(ЧтениеJSON);
		ЧтениеJSON.Закрыть();

		Нужные = Новый Массив;
		Если ТипЗнч(ДанныеТребований) = Тип("Структура") И ДанныеТребований.Свойство("items") Тогда
			Нужные = ДанныеТребований.items;
		КонецЕсли;

		Пакет = Новый Массив;
		Для Каждого СтрокаНужна Из Нужные Цикл
			ИдНоменклатуры = "";
			ИдВидаЦен = "";
			Если ТипЗнч(СтрокаНужна) = Тип("Структура") Тогда
				Если СтрокаНужна.Свойство("id") Тогда
					ИдНоменклатуры = Строка(СтрокаНужна.id);
				КонецЕсли;
				Если СтрокаНужна.Свойство("priceTypeId") Тогда
					ИдВидаЦен = Строка(СтрокаНужна.priceTypeId);
				КонецЕсли;
			КонецЕсли;
			Если ПустаяСтрока(ИдНоменклатуры) Или ПустаяСтрока(ИдВидаЦен) Тогда
				Продолжить;
			КонецЕсли;

			Попытка
				НоменклатураСсылка = Справочники.Номенклатура.ПолучитьСсылку(
					Новый УникальныйИдентификатор(ИдНоменклатуры));
				ВидЦенСсылка = Справочники.ВидыЦен.ПолучитьСсылку(
					Новый УникальныйИдентификатор(ИдВидаЦен));
			Исключение
				Продолжить;
			КонецПопытки;

			ЦенаЗначение = 0;
			Попытка
				ЗапросЦены = Новый Запрос;
				ЗапросЦены.Текст =
				"ВЫБРАТЬ ПЕРВЫЕ 1
				|	Цены.Цена КАК Цена
				|ИЗ
				|	РегистрСведений.ЦеныНоменклатуры.СрезПоследних(
				|		,
				|		Номенклатура = &Номенклатура
				|			И ВидЦен = &ВидЦен) КАК Цены";
				ЗапросЦены.УстановитьПараметр("Номенклатура", НоменклатураСсылка);
				ЗапросЦены.УстановитьПараметр("ВидЦен", ВидЦенСсылка);
				ВыборкаЦены = ЗапросЦены.Выполнить().Выбрать();
				Если ВыборкаЦены.Следующий() Тогда
					ЦенаЗначение = ВыборкаЦены.Цена;
				КонецЕсли;
			Исключение
				Продолжить;
			КонецПопытки;

			Если ЦенаЗначение = 0 Тогда
				Продолжить;
			КонецЕсли;

			СтрокаПакета = Новый Структура;
			СтрокаПакета.Вставить("id", ИдНоменклатуры);
			СтрокаПакета.Вставить("priceTypeId", ИдВидаЦен);
			СтрокаПакета.Вставить("price", ЦенаЗначение);
			Пакет.Добавить(СтрокаПакета);
		КонецЦикла;

		Данные = Новый Структура;
		Данные.Вставить("items", Пакет);
		ЗаписьJSON = Новый ЗаписьJSON;
		ЗаписьJSON.УстановитьСтроку();
		ЗаписатьJSON(ЗаписьJSON, Данные);
		ТелоJSON = ЗаписьJSON.Закрыть();

		ЗапросHTTP = Новый HTTPЗапрос("/api/one-c/sale-prices");
		ЗапросHTTP.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
		ДобавитьЗаголовкиОбмена(ЗапросHTTP);
		ЗапросHTTP.УстановитьТелоИзСтроки(ТелоJSON, КодировкаТекста.UTF8);
		Ответ = Соединение.ОтправитьДляОбработки(ЗапросHTTP);

		Результат.КодСостояния = Ответ.КодСостояния;
		Результат.Количество = Пакет.Количество();
		Если Ответ.КодСостояния <> 200 Тогда
			Результат.Ошибка = Ответ.ПолучитьТелоКакСтроку();
		КонецЕсли;
		Возврат Результат;
	Исключение
		Результат.Ошибка = ПодробноеПредставлениеОшибки(ИнформацияОбОшибке());
		Возврат Результат;
	КонецПопытки;

КонецФункции
