&НаКлиенте
Процедура ОбработкаКоманды(ПараметрКоманды, ПараметрыВыполненияКоманды)
	
	// Важно: вызывать с клиента, НЕ через &НаСервере —
	// иначе 1С включает безопасный режим и блокирует HTTPСоединение.
	//
	// Одна кнопка: контрагенты (+ виды цен) и продажные цены матрицы.
	// В модуле ОтправитьКонтрагентыПредпросмотр уже вызывает
	// ОтправитьПродажныеЦеныВClover() после успешной выгрузки.
	
	Результат = Clover_ОбменКлиент.ОтправитьКонтрагентыПредпросмотр();
	
	Если Результат.КодСостояния = 200 Тогда
		Текст = "Контрагенты и цены переданы в Clover. Контрагентов: "
			+ Строка(Результат.Количество);
		КоличествоЦен = 0;
		Если Результат.Свойство("КоличествоЦен") Тогда
			КоличествоЦен = Результат.КоличествоЦен;
		КонецЕсли;
		Текст = Текст + ". Цен: " + Строка(КоличествоЦен);
		ОшибкаЦен = "";
		Если Результат.Свойство("ОшибкаЦен") Тогда
			ОшибкаЦен = СокрЛП(Строка(Результат.ОшибкаЦен));
		КонецЕсли;
		Если ОшибкаЦен <> "" Тогда
			Текст = Текст + ". Ошибка цен: " + ОшибкаЦен;
		КонецЕсли;
		Сообщить(Текст);
	ИначеЕсли Результат.КодСостояния = -1 Тогда
		ТекстОшибки = "";
		Если Результат.Свойство("Ошибка") Тогда
			ТекстОшибки = Строка(Результат.Ошибка);
		КонецЕсли;
		Если ТекстОшибки = "" Тогда
			Сообщить("Не удалось связаться с Clover (контрагенты и цены). Проверьте, что API запущен.");
		Иначе
			Сообщить("Ошибка выгрузки контрагентов и цен: " + ТекстОшибки);
		КонецЕсли;
	Иначе
		Сообщить("Clover ответил кодом " + Строка(Результат.КодСостояния)
			+ ". Контрагентов в пакете: " + Строка(Результат.Количество));
	КонецЕсли;
	
КонецПроцедуры
