Функция ОценитьИсточникЦены(ТекстИсточника) Экспорт
	Текст = НРег(Строка(ТекстИсточника));
	Оценка = 0;

	Если Найти(Текст, "закуп") > 0 Тогда
		Оценка = Оценка + 100;
	КонецЕсли;
	Если Найти(Текст, "постав") > 0 Тогда
		Оценка = Оценка + 80;
	КонецЕсли;
	Если Найти(Текст, "контрагент") > 0 Тогда
		Оценка = Оценка + 50;
	КонецЕсли;
	Если Найти(Текст, "себесто") > 0 Тогда
		Оценка = Оценка + 40;
	КонецЕсли;
	Если Найти(Текст, "цен") > 0 Тогда
		Оценка = Оценка + 10;
	КонецЕсли;

	Если Найти(Текст, "рознич") > 0 Тогда
		Оценка = Оценка - 120;
	КонецЕсли;
	Если Найти(Текст, "оптов") > 0 Тогда
		Оценка = Оценка - 100;
	КонецЕсли;
	Если Найти(Текст, "продаж") > 0 Тогда
		Оценка = Оценка - 100;
	КонецЕсли;

	Возврат Оценка;

Функция НайтиЦенуВРегистре(
	ИмяРегистра,
	ИмяПоляНоменклатуры,
	ИмяПоляЦены,
	ИмяПоляВидаЦен,
	Номенклатура,
	ОценкаРегистра,
	ИспользоватьПериод) Экспорт

	Результат = НовыйРезультатЦены();

	ТекстЗапроса =
		"ВЫБРАТЬ ПЕРВЫЕ 50 "
		+ "Регистр." + ИмяПоляЦены + " КАК Цена";

	Если ИмяПоляВидаЦен <> "" Тогда
		ТекстЗапроса = ТекстЗапроса
			+ ", Регистр." + ИмяПоляВидаЦен + " КАК ВидЦены";
	КонецЕсли;

	Если ИспользоватьПериод Тогда
		ТекстЗапроса = ТекстЗапроса + ", Регистр.Период КАК ДатаЦены";
	КонецЕсли;

	ТекстЗапроса = ТекстЗапроса
		+ " ИЗ РегистрСведений." + ИмяРегистра + " КАК Регистр"
		+ " ГДЕ Регистр." + ИмяПоляНоменклатуры + " = &Номенклатура"
		+ " И Регистр." + ИмяПоляЦены + " > 0";

	Если ИспользоватьПериод Тогда
		ТекстЗапроса = ТекстЗапроса
			+ " УПОРЯДОЧИТЬ ПО Регистр.Период УБЫВ";
	КонецЕсли;

	ЗапросЦены = Новый Запрос(ТекстЗапроса);
	ЗапросЦены.УстановитьПараметр("Номенклатура", Номенклатура);
	ВыборкаЦены = ЗапросЦены.Выполнить().Выбрать();

	Пока ВыборкаЦены.Следующий() Цикл
		ТекстВидаЦен = "";
		Если ИмяПоляВидаЦен <> "" Тогда
			ТекстВидаЦен = Строка(ВыборкаЦены.ВидЦены);
		КонецЕсли;

		Оценка = ОценкаРегистра + ОценитьИсточникЦены(ТекстВидаЦен);
		Если ВыборкаЦены.Цена > 0
			И Оценка > Результат.Оценка
			И Оценка >= 20 Тогда

			Результат.Успех = Истина;
			Результат.Цена = ВыборкаЦены.Цена;
			Результат.Источник = ИмяРегистра;

			Если ТекстВидаЦен <> "" Тогда
				Результат.Источник = Результат.Источник
					+ " / " + ТекстВидаЦен;
			КонецЕсли;

			Результат.Оценка = Оценка;
		КонецЕсли;
	КонецЦикла;

	Возврат Результат;

Функция ПолучитьЦенуИзРегистров(Номенклатура) Экспорт
	ЛучшийРезультат = НовыйРезультатЦены();

	Для Каждого МетаРегистр Из Метаданные.РегистрыСведений Цикл
		ТекстРегистра = НРег(
			МетаРегистр.Имя + " " + Строка(МетаРегистр.Синоним));

		ОценкаРегистра = ОценитьИсточникЦены(ТекстРегистра);
		Если ОценкаРегистра < 10 Тогда
			Продолжить;
		КонецЕсли;

		ИмяПоляНоменклатуры = "";
		ИмяПоляВидаЦен = "";
		ИмяПоляЦены = "";
		ЛучшаяОценкаПоляЦены = -1;

		Для Каждого МетаПоле Из МетаРегистр.Измерения Цикл
			ТекстПоля = НРег(
				МетаПоле.Имя + " " + Строка(МетаПоле.Синоним));

			Если ИмяПоляНоменклатуры = ""
				И Найти(ТекстПоля, "номенклатур") > 0
				И Найти(ТекстПоля, "поставщик") = 0 Тогда

				ИмяПоляНоменклатуры = МетаПоле.Имя;
			КонецЕсли;

			Если ИмяПоляВидаЦен = ""
				И (Найти(ТекстПоля, "вид цен") > 0
					ИЛИ Найти(ТекстПоля, "видцен") > 0
					ИЛИ Найти(ТекстПоля, "тип цен") > 0
					ИЛИ Найти(ТекстПоля, "типцен") > 0) Тогда

				ИмяПоляВидаЦен = МетаПоле.Имя;
			КонецЕсли;
		КонецЦикла;

		Для Каждого МетаПоле Из МетаРегистр.Ресурсы Цикл
			ТекстПоля = НРег(
				МетаПоле.Имя + " " + Строка(МетаПоле.Синоним));

			ОценкаПоляЦены = -1;
			Если НРег(МетаПоле.Имя) = "цена" Тогда
				ОценкаПоляЦены = 100;
			ИначеЕсли Найти(ТекстПоля, "закуп") > 0
				И Найти(ТекстПоля, "цен") > 0 Тогда

				ОценкаПоляЦены = 90;
			ИначеЕсли Найти(ТекстПоля, "постав") > 0
				И Найти(ТекстПоля, "цен") > 0 Тогда

				ОценкаПоляЦены = 80;
			ИначеЕсли Найти(ТекстПоля, "цен") > 0
				И Найти(ТекстПоля, "процент") = 0
				И Найти(ТекстПоля, "валют") = 0 Тогда

				ОценкаПоляЦены = 20;
			КонецЕсли;

			Если ОценкаПоляЦены > ЛучшаяОценкаПоляЦены Тогда
				ЛучшаяОценкаПоляЦены = ОценкаПоляЦены;
				ИмяПоляЦены = МетаПоле.Имя;
			КонецЕсли;
		КонецЦикла;

		Если ИмяПоляЦены = "" Тогда
			Для Каждого МетаПоле Из МетаРегистр.Реквизиты Цикл
				ТекстПоля = НРег(
					МетаПоле.Имя + " " + Строка(МетаПоле.Синоним));

				ОценкаПоляЦены = -1;
				Если НРег(МетаПоле.Имя) = "цена" Тогда
					ОценкаПоляЦены = 100;
				ИначеЕсли Найти(ТекстПоля, "закуп") > 0
					И Найти(ТекстПоля, "цен") > 0 Тогда

					ОценкаПоляЦены = 90;
				ИначеЕсли Найти(ТекстПоля, "постав") > 0
					И Найти(ТекстПоля, "цен") > 0 Тогда

					ОценкаПоляЦены = 80;
				ИначеЕсли Найти(ТекстПоля, "цен") > 0
					И Найти(ТекстПоля, "процент") = 0
					И Найти(ТекстПоля, "валют") = 0 Тогда

					ОценкаПоляЦены = 20;
				КонецЕсли;

				Если ОценкаПоляЦены > ЛучшаяОценкаПоляЦены Тогда
					ЛучшаяОценкаПоляЦены = ОценкаПоляЦены;
					ИмяПоляЦены = МетаПоле.Имя;
				КонецЕсли;
			КонецЦикла;
		КонецЕсли;

		Если ИмяПоляНоменклатуры = "" Или ИмяПоляЦены = "" Тогда
			Продолжить;
		КонецЕсли;

		РезультатРегистра = НовыйРезультатЦены();

		Попытка
			РезультатРегистра = НайтиЦенуВРегистре(
				МетаРегистр.Имя,
				ИмяПоляНоменклатуры,
				ИмяПоляЦены,
				ИмяПоляВидаЦен,
				Номенклатура,
				ОценкаРегистра,
				Истина);
		Исключение
			Попытка
				РезультатРегистра = НайтиЦенуВРегистре(
					МетаРегистр.Имя,
					ИмяПоляНоменклатуры,
					ИмяПоляЦены,
					ИмяПоляВидаЦен,
					Номенклатура,
					ОценкаРегистра,
					Ложь);
			Исключение
				РезультатРегистра.Ошибка = ОписаниеОшибки();
			КонецПопытки;
		КонецПопытки;

		Если РезультатРегистра.Успех
			И РезультатРегистра.Оценка > ЛучшийРезультат.Оценка Тогда

			ЛучшийРезультат = РезультатРегистра;
		КонецЕсли;
	КонецЦикла;

	Возврат ЛучшийРезультат;

	Ид = СокрЛП(Строка(Идентификатор));

Функция ОбновитьЗакупочныеЦеныСледующегоЗаказа() Экспорт
	Результат = Новый Структура;
	Результат.Вставить("Успех", Ложь);
	Результат.Вставить("Ошибка", "");
	Результат.Вставить("Количество", 0);

	Попытка
		Соединение = Новый HTTPСоединение("192.168.155.15", 4100);

		ЗапросТребований =
			Новый HTTPЗапрос(
				"/api/one-c/purchase-price-request?scope=next-order");
		ДобавитьЗаголовкиОбмена(ЗапросТребований);

		ОтветТребований = Соединение.Получить(ЗапросТребований);
		ТекстТребований =
			ОтветТребований.ПолучитьТелоКакСтроку();

		Если ОтветТребований.КодСостояния <> 200 Тогда
			ВызватьИсключение
				"Не получен запрос закупочных цен. HTTP "
				+ Строка(ОтветТребований.КодСостояния)
				+ ": " + ТекстТребований;
		КонецЕсли;

		ЧтениеJSON = Новый ЧтениеJSON;
		ЧтениеJSON.УстановитьСтроку(ТекстТребований);
		ДанныеТребований = ПрочитатьJSON(ЧтениеJSON);
		ЧтениеJSON.Закрыть();

		ТребуемыеТовары = ДанныеТребований.items;
		Если ТребуемыеТовары.Количество() = 0 Тогда
			Результат.Успех = Истина;
			Возврат Результат;
		КонецЕсли;

		ЦеныДляОтправки = Новый Массив;

		Для Каждого Требование Из ТребуемыеТовары Цикл
			Номенклатура =
				НайтиНоменклатуруДляОбмена(
					Требование.id,
					Требование.code);

			Если Номенклатура.Пустая() Тогда
				ВызватьИсключение
					"В 1С не найдена номенклатура: "
					+ Строка(Требование.name)
					+ " (" + Строка(Требование.code) + ").";
			КонецЕсли;

			Цена = ПолучитьАктуальнуюЗакупочнуюЦену(Номенклатура);
			Если Не Цена.Успех Тогда
				ВызватьИсключение
					"Для товара """ + Строка(Требование.name)
					+ """ не найдена закупочная цена. "
					+ Цена.Ошибка;
			КонецЕсли;

			ЭлементЦены = Новый Структура;
			ЭлементЦены.Вставить("id", Строка(Требование.id));
			ЭлементЦены.Вставить("code", Строка(Требование.code));
			ЭлементЦены.Вставить("name", Строка(Требование.name));
			ЭлементЦены.Вставить("purchasePrice", Цена.Цена);
			ЭлементЦены.Вставить("purchasePriceUnit", "piece");
			ЭлементЦены.Вставить("purchasePriceSource", Цена.Источник);
			ЦеныДляОтправки.Добавить(ЭлементЦены);
		КонецЦикла;

		ДанныеЦен = Новый Структура;
		ДанныеЦен.Вставить("database", "TEST");
		ДанныеЦен.Вставить("items", ЦеныДляОтправки);

		ЗаписьJSON = Новый ЗаписьJSON;
		ЗаписьJSON.УстановитьСтроку();
		ЗаписатьJSON(ЗаписьJSON, ДанныеЦен);
		ТелоЦен = ЗаписьJSON.Закрыть();

		ЗапросЦен = Новый HTTPЗапрос("/api/one-c/purchase-prices");
		ЗапросЦен.Заголовки.Вставить(
			"Content-Type",
			"application/json; charset=utf-8");
		ДобавитьЗаголовкиОбмена(ЗапросЦен);
		ЗапросЦен.УстановитьТелоИзСтроки(
			ТелоЦен,
			КодировкаТекста.UTF8);

		ОтветЦен = Соединение.ОтправитьДляОбработки(ЗапросЦен);
		ТекстОтветаЦен = ОтветЦен.ПолучитьТелоКакСтроку();

		Если ОтветЦен.КодСостояния <> 200 Тогда
			ВызватьИсключение
				"Clover не принял закупочные цены. HTTP "
				+ Строка(ОтветЦен.КодСостояния)
				+ ": " + ТекстОтветаЦен;
		КонецЕсли;

		ЧтениеJSON = Новый ЧтениеJSON;
		ЧтениеJSON.УстановитьСтроку(ТекстОтветаЦен);
		ДанныеОтветаЦен = ПрочитатьJSON(ЧтениеJSON);
		ЧтениеJSON.Закрыть();

		Если ДанныеОтветаЦен.accepted
			<> ЦеныДляОтправки.Количество() Тогда

			ВызватьИсключение
				"Clover принял не все закупочные цены: "
				+ ТекстОтветаЦен;
		КонецЕсли;

		Результат.Успех = Истина;
		Результат.Количество = ЦеныДляОтправки.Количество();

	Исключение
		Результат.Ошибка = ОписаниеОшибки();
	КонецПопытки;

	Возврат Результат;

	// менеджер выставит его вручную в 1С; тогда Clover поставит клиенту «Принят».

		Возврат;

	Попытка
		МетаРеквизит = Заказ.Метаданные().Реквизиты.Найти(ИмяСостояния);

Процедура ЗаполнитьЕдиницуКоличествоИЦену(
	СтрокаЗапаса,
	МетаСтроки,
	Номенклатура,
	Позиция) Экспорт

	// Правило Clover: в документ 1С всегда базовая единица «шт».
	// Пачка/упаковка из Clover пересчитывается в штуки с сохранением суммы.

	ИменаЕдиницы = Новый Массив;
	ИменаЕдиницы.Добавить("ЕдиницаИзмерения");
	ИменаЕдиницы.Добавить("Единица");
	ИмяПоляЕдиницы = НайтиИмяРеквизитаПоИменам(
		МетаСтроки, ИменаЕдиницы);
	Если ИмяПоляЕдиницы = "" Тогда
		КлючиЕдиницы = Новый Массив;
		КлючиЕдиницы.Добавить("единиц");
		ИмяПоляЕдиницы = НайтиИмяРеквизитаПоКлючам(
			МетаСтроки, КлючиЕдиницы);
	КонецЕсли;

	ИменаКоэффициента = Новый Массив;
	ИменаКоэффициента.Добавить("Коэффициент");
	ИменаКоэффициента.Добавить("Кратность");
	ИмяПоляКоэффициента = НайтиИмяРеквизитаПоИменам(
		МетаСтроки, ИменаКоэффициента);

	БазоваяЕдиница = ПолучитьБазовуюЕдиницуНоменклатуры(Номенклатура);
	Если БазоваяЕдиница <> Неопределено
		И ИмяПоляЕдиницы <> "" Тогда
		СтрокаЗапаса[ИмяПоляЕдиницы] = БазоваяЕдиница;
	КонецЕсли;
	Если ИмяПоляКоэффициента <> "" Тогда
		СтрокаЗапаса[ИмяПоляКоэффициента] = 1;
	КонецЕсли;

	Количество = Число(ПолеПозицииClover(Позиция, "quantity"));
	Цена = Число(ПолеПозицииClover(Позиция, "price"));
	СуммаСтроки = Количество * Цена;

	КодЕдиницы = СокрЛП(Строка(ПолеПозицииClover(Позиция, "unit")));
	Если КодЕдиницы = "" Тогда
		КодЕдиницы = "piece";
	КонецЕсли;

	Множитель = 1;
	ЗначениеМножителя = ПолеПозицииClover(Позиция, "multiplier");
	Если Число(ЗначениеМножителя) > 0 Тогда
		Множитель = Число(ЗначениеМножителя);
	КонецЕсли;

	ЗначениеВсегоШт = ПолеПозицииClover(Позиция, "totalPieces");
	Если Число(ЗначениеВсегоШт) > 0 Тогда
		Количество = Число(ЗначениеВсегоШт);
	ИначеЕсли КодЕдиницы <> "piece" И Множитель > 1 Тогда
		Количество = Количество * Множитель;
	КонецЕсли;

	Если Количество > 0 Тогда
		Цена = СуммаСтроки / Количество;
	КонецЕсли;

	СтрокаЗапаса.Количество = Количество;
	СтрокаЗапаса.Цена = Цена;
	СтрокаЗапаса.Сумма = Количество * Цена;
	СтрокаЗапаса.Всего = СтрокаЗапаса.Сумма;
	УстановитьНулевуюРучнуюСкидку(СтрокаЗапаса, МетаСтроки);
КонецПроцедуры

Процедура СообщитьОПринятииВClover(Заказ) Экспорт
	// Ручная смена состояния в 1С → у клиента Clover статус «Принят».

	Если ТипЗнч(Заказ) <> Тип("ДокументОбъект.ЗаказПокупателя") Тогда
		Возврат;
	КонецЕсли;

	Если Заказ.ДополнительныеСвойства.Свойство("CloverSkipAcceptedNotify") Тогда
		Возврат;
	КонецЕсли;

	НомерClover = ИзвлечьНомерЗаказаCloverИзКомментария(Заказ.Комментарий);
	Если НомерClover = "" Тогда
		Возврат;
	КонецЕсли;

	ИмяСостояния = ИмяРеквизитаСостоянияЗаказа(Заказ);
	Если ИмяСостояния = "" Тогда
		Возврат;
	КонецЕсли;
	Если Не ЗначениеЗаполнено(Заказ[ИмяСостояния]) Тогда
		Возврат;
	КонецЕсли;

	СостояниеТекст = СокрЛП(Строка(Заказ[ИмяСостояния]));
	Если СостояниеТекст = "" Тогда
		Возврат;
	КонецЕсли;

	// Только если состояние реально изменилось (в т.ч. с пустого на любое).
	ПредыдущееСостояние = Неопределено;
	Если Заказ.ДополнительныеСвойства.Свойство(
		"CloverPrevState", ПредыдущееСостояние) Тогда
		Если ЗначениеЗаполнено(ПредыдущееСостояние)
			И ПредыдущееСостояние = Заказ[ИмяСостояния] Тогда
			Возврат;
		КонецЕсли;
		Если Не ЗначениеЗаполнено(ПредыдущееСостояние)
			И Не ЗначениеЗаполнено(Заказ[ИмяСостояния]) Тогда
			Возврат;
		КонецЕсли;
	КонецЕсли;

	СообщитьОПринятииПоРеквизитам(
		НомерClover,
		Строка(Заказ.Номер),
		СостояниеТекст);

Процедура ЗаполнитьЗапасыЗаказаИзClover(
	Заказ,
	ЗаказОбразец,
	ДанныеЗаказа) Экспорт

	МетаСтроки = Заказ.Метаданные().ТабличныеЧасти.Найти("Запасы");
	Если МетаСтроки = Неопределено Тогда
		ВызватьИсключение "В заказе 1С не найдена табличная часть Запасы.";
	КонецЕсли;

	СтрокаОбразца = Неопределено;
	Если ЗаказОбразец <> Неопределено
		И ЗаказОбразец.Запасы.Количество() > 0 Тогда
		СтрокаОбразца = ЗаказОбразец.Запасы[0];
	КонецЕсли;

	ТоварыЗаказа = МассивТоваровЗаказаClover(ДанныеЗаказа);
	Если ТоварыЗаказа.Количество() = 0 Тогда
		ВызватьИсключение "В заказе Clover пустой список товаров.";
	КонецЕсли;

	Заказ.Запасы.Очистить();

	Для Каждого Позиция Из ТоварыЗаказа Цикл
		ИдНоменклатуры = ПолеПозицииClover(Позиция, "id");
		КодНоменклатуры = ПолеПозицииClover(Позиция, "code");
		Номенклатура = НайтиНоменклатуруДляОбмена(
			ИдНоменклатуры, КодНоменклатуры);
		Если Номенклатура.Пустая() Тогда
			ВызватьИсключение
				"Не найдена номенклатура с кодом "
				+ Строка(КодНоменклатуры);
		КонецЕсли;

		СтрокаЗапаса = Заказ.Запасы.Добавить();
		Если СтрокаОбразца <> Неопределено Тогда
			КопироватьРеквизитСтрокиЕслиЕсть(
				СтрокаЗапаса, СтрокаОбразца, МетаСтроки, "Склад");
			КопироватьРеквизитСтрокиЕслиЕсть(
				СтрокаЗапаса, СтрокаОбразца, МетаСтроки, "СтавкаНДС");
			КопироватьРеквизитСтрокиЕслиЕсть(
				СтрокаЗапаса, СтрокаОбразца, МетаСтроки, "НалогообложениеНДС");
		КонецЕсли;

		СтрокаЗапаса.Номенклатура = Номенклатура;
		ЗаполнитьЕдиницуКоличествоИЦену(
			СтрокаЗапаса, МетаСтроки, Номенклатура, Позиция);

		ИмяСуммыНДС = "";
		ИменаСуммыНДС = Новый Массив;
		ИменаСуммыНДС.Добавить("СуммаНДС");
		ИмяСуммыНДС = НайтиИмяРеквизитаПоИменам(
			МетаСтроки, ИменаСуммыНДС);
		Если ИмяСуммыНДС <> "" Тогда
			Попытка
				СтрокаЗапаса[ИмяСуммыНДС] = 0;
			Исключение
			КонецПопытки;
		КонецЕсли;
	КонецЦикла;

	Если Заказ.Запасы.Количество() = 0 Тогда
		ВызватьИсключение
			"Товары Clover не попали в табличную часть заказа 1С.";
	КонецЕсли;

Функция СоздатьОдинТестовыйЗаказ()

	Результат = Новый Структура;
	Результат.Вставить("Успех", Ложь);

		Результат.Ошибка = РезультатЦен.Ошибка;
		Возврат Результат;
	КонецЕсли;

		Соединение = Новый HTTPСоединение("192.168.155.15", 4100);
		ЗапросHTTP = Новый HTTPЗапрос("/api/one-c/test-order");
		ДобавитьЗаголовкиОбмена(ЗапросHTTP);
		Ответ = Соединение.Получить(ЗапросHTTP);
		ТекстОтвета = Ответ.ПолучитьТелоКакСтроку();

		Если Ответ.КодСостояния = 404 Тогда
			// Пустая очередь — штатная ситуация, короткое сообщение без HTTP/JSON.
			Результат.Ошибка = "Нет новых заказов";
			Возврат Результат;
		КонецЕсли;

		Если Ответ.КодСостояния <> 200 Тогда
			ВызватьИсключение
				"Clover вернул HTTP-код "
				+ Строка(Ответ.КодСостояния)
				+ ": " + ТекстОтвета;
		КонецЕсли;

		ЧтениеJSON = Новый ЧтениеJSON;
		ЧтениеJSON.УстановитьСтроку(ТекстОтвета);
		ДанныеОтвета = ПрочитатьJSON(ЧтениеJSON);
		ЧтениеJSON.Закрыть();
		ДанныеЗаказа = ДанныеОтвета.order;

		// Ищем повтор по короткому комментарию, без служебного [CLOVER:uuid].
		Маркер = "Заказ Clover № " + Строка(ДанныеЗаказа.number);
		Заказ = НайтиРанееСозданныйЗаказ(Маркер);

		Если Заказ = Неопределено Тогда
			ЗаказОбразец = ПолучитьЗаказОбразец();

				ВызватьИсключение
					"Не найден рабочий заказ-образец №3265. "

					+ "организации, склада, валюты и НДС.";
			КонецЕсли;

			Заказ = Документы.ЗаказПокупателя.СоздатьДокумент();
			Заказ.ВидОперации =
				Перечисления.ВидыОперацийЗаказПокупателя.ЗаказНаПродажу;
			Заказ.Дата = ТекущаяДата();
			ЗаполнитьШапкуПоОбразцу(Заказ, ЗаказОбразец);

			Контрагент = НайтиКонтрагентаПоДанным(ДанныеЗаказа.customer);
			Если Контрагент.Пустая() Тогда
				ВызватьИсключение
					"Контрагент не найден: "
					+ Строка(ДанныеЗаказа.customer.name);
			КонецЕсли;
			Заказ.Контрагент = Контрагент;

			ИменаДоговора = Новый Массив;
			ИменаДоговора.Добавить("Договор");
			ИменаДоговора.Добавить("ДоговорКонтрагента");
			ИмяДоговора = НайтиИмяРеквизитаПоИменам(
				Заказ.Метаданные(), ИменаДоговора);

			Договор = НайтиОсновнойДоговор(Контрагент, МетаДоговора);
			Если Договор = Неопределено Тогда
				ВызватьИсключение
					"Для покупателя " + Строка(Контрагент)
					+ " не найден договор с наименованием «Основной договор».";
			КонецЕсли;
			Заказ[ИмяДоговора] = Договор;

			ДатаОтгрузки = Неопределено;
			Если ДанныеЗаказа.Свойство("deliveryDate", ЗначениеДаты) Тогда
				ДатаОтгрузки = ПреобразоватьДатуClover(ЗначениеДаты);
			КонецЕсли;
			Если ДатаОтгрузки = Неопределено Тогда
				ВызватьИсключение
					"В заказе Clover не указана корректная дата доставки.";
			КонецЕсли;

			АдресДоставки = "";
			Если ДанныеЗаказа.Свойство("address", ЗначениеАдреса) Тогда
				АдресДоставки = СокрЛП(Строка(ЗначениеАдреса));
			КонецЕсли;
			ИменаАдреса = Новый Массив;
			ИменаАдреса.Добавить("АдресДоставки");
			ИменаАдреса.Добавить("Адрес");
			ИмяАдреса = НайтиИмяРеквизитаПоИменам(
				Заказ.Метаданные(), ИменаАдреса);
			Если ИмяАдреса <> "" И АдресДоставки <> "" Тогда
				Попытка
					Заказ[ИмяАдреса] = АдресДоставки;
				Исключение
				КонецПопытки;
			КонецЕсли;

			Если Заказ.Метаданные().Реквизиты.Найти("КонтактныйТелефон") <> Неопределено
				И ДанныеЗаказа.customer.Свойство("phone", ЗначениеТелефона) Тогда
				Попытка
					Заказ.КонтактныйТелефон = Строка(ЗначениеТелефона);
				Исключение
				КонецПопытки;
			КонецЕсли;

			Если Заказ.Метаданные().Реквизиты.Найти("КонтактноеЛицо") <> Неопределено
				И ДанныеЗаказа.customer.Свойство("contactName", ЗначениеКонтакта)
				И СокрЛП(Строка(ЗначениеКонтакта)) <> "" Тогда
				Попытка
					МетаКонтакта = Заказ.Метаданные().Реквизиты.Найти("КонтактноеЛицо");
					Если Найти(НРег(Строка(МетаКонтакта.Тип)), "строка") > 0 Тогда
						Заказ.КонтактноеЛицо = Строка(ЗначениеКонтакта);
					КонецЕсли;
				Исключение
				КонецПопытки;
			КонецЕсли;

			// В комментарии документа только короткий текст.
			// Адрес пишется в реквизит адреса выше, UUID остаётся в ACK по id.
			Комментарий = "Заказ Clover № " + Строка(ДанныеЗаказа.number);
			ЗначениеКомментария = "";
			Если ДанныеЗаказа.Свойство("comment", ЗначениеКомментария)
				И СокрЛП(Строка(ЗначениеКомментария)) <> "" Тогда
				Комментарий = Строка(ЗначениеКомментария);
			ИначеЕсли ДанныеЗаказа.Свойство("clientComment", ЗначениеКомментария)
				И СокрЛП(Строка(ЗначениеКомментария)) <> "" Тогда
				Комментарий = Комментарий
					+ Символы.ПС + "Комментарий"
					+ Символы.ПС + Строка(ЗначениеКомментария);
			КонецЕсли;
			Заказ.Комментарий = Комментарий;

			ЗаполнитьЗапасыЗаказаИзClover(Заказ, ЗаказОбразец, ДанныеЗаказа);

			// Дата отгрузки — только в шапке (поле под «Договор»), не в строках.
			УстановитьДатуОтгрузкиВШапке(Заказ, ДатаОтгрузки);

			// Состояние намеренно пустое: менеджер выставит вручную в 1С.
			ОчиститьСостояниеЗаказа(Заказ);

			Заказ.ДополнительныеСвойства.Вставить(
				"CloverSkipAcceptedNotify", Истина);
			Заказ.Записать();
		ИначеЕсли Заказ.Запасы.Количество() = 0 Тогда
			ЗаказОбразец = ПолучитьЗаказОбразец();
			ЗаполнитьЗапасыЗаказаИзClover(Заказ, ЗаказОбразец, ДанныеЗаказа);
			Заказ.ДополнительныеСвойства.Вставить(
				"CloverSkipAcceptedNotify", Истина);
			Заказ.Записать();
		КонецЕсли;

		Если Заказ.Запасы.Количество() = 0 Тогда
			ВызватьИсключение
				"Заказ 1С создан без товаров. Строки не подтверждены.";
		КонецЕсли;

		// Перед ACK убеждаемся, что номер относится именно к документу
		// с маркером текущего заказа Clover.
		Если Найти(Строка(Заказ.Комментарий), Маркер) = 0 Тогда
			ВызватьИсключение
				"Защита Clover: документ 1С не содержит маркер текущего заказа. "
				+ "Подтверждение не отправлено.";
		КонецЕсли;
		Если Не ЗначениеЗаполнено(Заказ.Номер) Тогда
			ВызватьИсключение
				"Защита Clover: 1С не присвоила номер новому документу. "
				+ "Подтверждение не отправлено.";
		КонецЕсли;

		ПодтвердитьЗаказВClover(
			Соединение,
			ДанныеЗаказа.id,
			ДанныеЗаказа.number,
			Заказ.Номер);
		Результат.Успех = Истина;
		Результат.Номер = Заказ.Номер;

	Исключение
		Результат.Ошибка = ОписаниеОшибки();
	КонецПопытки;

	Возврат Результат;
КонецФункции

// Забирает ВСЕ заказы из очереди (каждый — отдельный claim + документ + ACK).
Функция СоздатьВсеТестовыеЗаказы() Экспорт

	Результат = Новый Структура;
	Результат.Вставить("Успех", Ложь);
	Результат.Вставить("Количество", 0);
	Результат.Вставить("Номера", "");
	Результат.Вставить("Ошибка", "");

	МаксимумЗаказов = 100;
	СписокНомеров = Новый Массив;
	ПоследняяОшибка = "";

	Пока СписокНомеров.Количество() < МаксимумЗаказов Цикл

		Один = СоздатьОдинТестовыйЗаказ();

		Если Один.Успех Тогда
			СписокНомеров.Добавить(Строка(Один.Номер));
			Продолжить;
		КонецЕсли;

		ПоследняяОшибка = СокрЛП(Строка(Один.Ошибка));

		Если ПоследняяОшибка = "Нет новых заказов" Тогда
			Прервать;
		КонецЕсли;

		// Ошибка на одном заказе: уже принятые сохраняем, цикл останавливаем.
		Результат.Количество = СписокНомеров.Количество();
		Результат.Номера = СобратьНомераЧерезЗапятую(СписокНомеров);
		Если Результат.Количество > 0 Тогда
			Результат.Успех = Истина;
			Результат.Ошибка =
				"Принято заказов: " + Строка(Результат.Количество)
				+ " (" + Результат.Номера + "). Остановлено из-за ошибки: "
				+ ПоследняяОшибка;
		Иначе
			Результат.Ошибка = ПоследняяОшибка;
		КонецЕсли;
		Возврат Результат;

	КонецЦикла;

	Результат.Количество = СписокНомеров.Количество();
	Результат.Номера = СобратьНомераЧерезЗапятую(СписокНомеров);

	Если Результат.Количество = 0 Тогда
		Результат.Ошибка = "Нет новых заказов";
		Возврат Результат;
	КонецЕсли;

	Если СписокНомеров.Количество() >= МаксимумЗаказов Тогда
		Результат.Успех = Истина;
		Результат.Ошибка =
			"Достигнут лимит " + Строка(МаксимумЗаказов)
			+ " заказов за один клик. Принято: " + Результат.Номера
			+ ". Нажмите кнопку ещё раз, если в очереди остались заказы.";
		Возврат Результат;
	КонецЕсли;

	Результат.Успех = Истина;
	Возврат Результат;

КонецФункции

Функция СобратьНомераЧерезЗапятую(СписокНомеров)

	Текст = "";
	Для Каждого НомерДокумента Из СписокНомеров Цикл
		Если Текст <> "" Тогда
			Текст = Текст + ", ";
		КонецЕсли;
		Текст = Текст + Строка(НомерДокумента);
	КонецЦикла;
	Возврат Текст;

Функция ОтправитьПродажныеЦеныВClover() Экспорт

			СтрокаПакета.Вставить("priceTypeId", ИдВидаЦен);
